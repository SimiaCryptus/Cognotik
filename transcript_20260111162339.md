<script>
    // Tab Switching Logic
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.style.display = 'none');
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).style.display = 'block';
        event.target.classList.add('active');
    }

    // Simulator Logic
    function loadPreset(type) {
        if (type === 'quarterly') {
            document.getElementById('sim_topic').value = "Q3 Marketing Performance";
            document.getElementById('sim_type').value = "quarterly_review";
            document.getElementById('sim_metrics').value = "ROI, CAC, Conversion Rate, Traffic Growth";
            document.getElementById('sim_risk').checked = true;
        } else if (type === 'incident') {
            document.getElementById('sim_topic').value = "Production DB Latency Spike";
            document.getElementById('sim_type').value = "incident_report";
            document.getElementById('sim_metrics').value = "P99 Latency, Error Rate, Recovery Time";
            document.getElementById('sim_risk').checked = false;
        }
    }

    function runSimulation() {
        const topic = document.getElementById('sim_topic').value;
        const type = document.getElementById('sim_type').value;
        const metrics = document.getElementById('sim_metrics').value.split(',');
        const revisions = parseInt(document.getElementById('sim_revisions').value);
        const includeRisk = document.getElementById('sim_risk').checked;
        const includeRecs = document.getElementById('sim_recs').checked;

        const logsEl = document.getElementById('sim-logs');
        const resultEl = document.getElementById('sim-result');

        // Reset UI
        logsEl.style.display = 'block';
        logsEl.innerHTML = '';
        resultEl.innerHTML = '<div style="text-align:center; color: var(--text-dim); margin-top: 2rem;">Processing...</div>';

        const logs = [
            `[INFO] Initializing ReportGenerationTask...`,
            `[INFO] Context: ${type.toUpperCase()} on '${topic}'`,
            `[INFO] Analyzing ${metrics.length} provided metrics...`,
            `[INFO] Drafting Executive Summary...`,
            `[INFO] Drafting Detailed Analysis...`,
        ];

        if (includeRisk) logs.push(`[INFO] Running Risk Assessment module...`);
        if (includeRecs) logs.push(`[INFO] Generating Strategic Recommendations...`);
        
        for(let i=1; i<=revisions; i++) {
            logs.push(`[INFO] Revision Pass ${i}/${revisions}: Checking tone and clarity...`);
            logs.push(`[INFO] Revision Pass ${i}/${revisions}: Applied 3 corrections.`);
        }
        
        logs.push(`[SUCCESS] Report generation complete.`);

        let logIndex = 0;
        const logInterval = setInterval(() => {
            if (logIndex < logs.length) {
                const line = document.createElement('div');
                line.textContent = `> ${logs[logIndex]}`;
                line.style.marginBottom = '2px';
                if (logs[logIndex].includes('SUCCESS')) line.style.color = '#4ade80';
                else if (logs[logIndex].includes('INFO')) line.style.color = '#60a5fa';
                
                logsEl.appendChild(line);
                logsEl.scrollTop = logsEl.scrollHeight;
                logIndex++;
            } else {
                clearInterval(logInterval);
                renderFinalReport(topic, type, metrics, includeRisk, includeRecs);
            }
        }, 400);
    }

    function renderFinalReport(topic, type, metrics, includeRisk, includeRecs) {
        const resultEl = document.getElementById('sim-result');
        const date = new Date().toLocaleDateString();
        
        let html = `
            <h1>${topic}</h1>
            <p style="color:var(--text-dim); font-size: 0.9em;"><strong>Type:</strong> ${type.replace('_', ' ').toUpperCase()} | <strong>Date:</strong> ${date}</p>
            
            <h2>Executive Summary</h2>
            <p>This report outlines the key findings regarding <strong>${topic}</strong>. Analysis of the provided metrics indicates a significant shift in performance patterns. Overall efficiency has improved, though specific bottlenecks remain.</p>
            
            <h2>Metrics Analysis</h2>
            <ul>
                ${metrics.map(m => `<li><strong>${m.trim()}:</strong> Analyzed. Current trajectory shows positive alignment with quarterly targets.</li>`).join('')}
            </ul>
        `;

        if (includeRecs) {
            html += `
                <h2>Recommendations</h2>
                <ul>
                    <li><strong>Optimize Resource Allocation:</strong> Shift focus to high-impact areas identified in the metric analysis.</li>
                    <li><strong>Standardize Reporting:</strong> Implement this format for future ${type.replace('_', ' ')}s.</li>
                </ul>
            `;
        }

        if (includeRisk) {
            html += `
                <h2>Risk Assessment</h2>
                <p><strong>Operational Risk (Medium):</strong> Dependency on external APIs may cause latency fluctuations.</p>
                <p><strong>Mitigation:</strong> Implement caching strategies immediately.</p>
            `;
        }

        resultEl.innerHTML = html;
    }
