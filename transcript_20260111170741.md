    async function runSimulation() {
        const logWindow = document.getElementById('sim-logs');
        const query = document.getElementById('sim-query').value;
        const strategy = document.getElementById('sim-strategy').value;
        const maxPages = parseInt(document.getElementById('sim-pages').value);
        
        logWindow.innerHTML = '';
        
        const addLog = (msg, type = 'info') => {
            const div = document.createElement('div');
            div.className = 'log-entry';
            const time = new Date().toLocaleTimeString('en-US', { hour12: false });
            let colorClass = 'log-info';
            if (type === 'success') colorClass = 'log-success';
            if (type === 'warn') colorClass = 'log-warn';
            if (type === 'error') colorClass = 'log-error';
            
            div.innerHTML = `<span class="log-time">[${time}]</span> <span class="${colorClass}">${msg}</span>`;
            logWindow.appendChild(div);
            logWindow.scrollTop = logWindow.scrollHeight;
        };

        const wait = (ms) => new Promise(r => setTimeout(r, ms));

        addLog(`Initializing CrawlerAgentTask...`);
        await wait(600);
        addLog(`Config: Query="${query}", Strategy=${strategy}, MaxPages=${maxPages}`);
        await wait(600);
        addLog(`Seeding via GoogleProxy...`);
        await wait(1000);
        addLog(`Found ${Math.floor(Math.random() * 5) + 3} seed URLs.`, 'success');

        const domains = ['techcrunch.com', 'wired.com', 'github.com', 'stackoverflow.com', 'medium.com'];

        for (let i = 1; i <= maxPages; i++) {
            const domain = domains[Math.floor(Math.random() * domains.length)];
            const url = `https://${domain}/article/${Math.floor(Math.random() * 1000)}`;
            
            addLog(`Processing Page ${i}: ${url}...`);
            await wait(Math.random() * 800 + 400);
            
            if (Math.random() > 0.8) {
                addLog(`Fetch failed: 403 Forbidden (Anti-bot detected)`, 'warn');
            } else {

                await wait(400);
                
                if (Math.random() > 0.5) {

                    addLog(`Extracted ${newLinks} new links. Added to queue.`, 'success');
                }
            }
        }

        await wait(500);
        addLog(`Crawling completed. Generating final report...`);
