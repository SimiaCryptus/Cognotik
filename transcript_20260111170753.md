<script>
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.style.display = 'none');
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).style.display = 'block';
        event.target.classList.add('active');
    }

    // Simulator Logic
    function runSimulation() {
        const desc = document.getElementById('sim-desc').value;
        const file = document.getElementById('sim-file').value;
        const btn = document.querySelector('.btn-run');
        const results = document.getElementById('sim-results');
        const logs = document.getElementById('sim-logs');
        const canvas = document.getElementById('preview-canvas');
        const ctx = canvas.getContext('2d');

        if (!desc || !file) {
            alert("Please provide a description and filename.");
            return;
        }

        btn.disabled = true;
        btn.textContent = "Generating...";
        results.style.display = 'flex';
        logs.innerHTML = '';
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        
        // Draw placeholder background
        ctx.fillStyle = '#111';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        const steps = [
            { msg: "Initializing ImageGenerator Agent...", delay: 500 },
            { msg: "Analyzing task description and context...", delay: 1200 },
            { msg: "Optimizing prompt for DALL-E...", delay: 2000 },
            { msg: "Sending request to image model...", delay: 3000 },
            { msg: "Receiving image data...", delay: 4500 },
            { msg: `Saving image to ${file}...`, delay: 5000 },
            { msg: "Task Completed Successfully.", delay: 5500 }
        ];

        let currentStep = 0;

        function addLog(msg) {
            const div = document.createElement('div');
            div.className = 'log-entry';
            div.textContent = `> ${msg}`;
            logs.appendChild(div);
            logs.scrollTop = logs.scrollHeight;
        }

        function drawGenerativeArt() {
            // Simple generative art to simulate result
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            const colors = ['#d946ef', '#8b5cf6', '#06b6d4', '#3b82f6', '#10b981'];
            
            for(let i=0; i<50; i++) {
                ctx.fillStyle = colors[Math.floor(Math.random() * colors.length)];
                ctx.globalAlpha = Math.random() * 0.5;
                const x = Math.random() * canvas.width;
                const y = Math.random() * canvas.height;
                const w = Math.random() * 150;
                const h = Math.random() * 150;
                ctx.fillRect(x, y, w, h);
            }
            ctx.globalAlpha = 1.0;
            ctx.fillStyle = '#fff';
            ctx.font = '16px Inter';
            ctx.fillText("AI Generated Preview: " + file, 20, 30);
        }

        function runStep() {
            if (currentStep >= steps.length) {
                btn.disabled = false;
                btn.textContent = "Run Generation Task";
                drawGenerativeArt();
                return;
            }

            const step = steps[currentStep];
            setTimeout(() => {
                addLog(step.msg);
                currentStep++;
                runStep();
            }, step.delay - (currentStep > 0 ? steps[currentStep-1].delay : 0));
        }

        runStep();
    }
</script>
