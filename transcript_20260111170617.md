    function switchTab(tabId, event) {
        document.querySelectorAll('.tab-pane').forEach(el => el.style.display = 'none');
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).style.display = 'block';
        if(event) event.target.classList.add('active');
    }

    // Simulator Logic
    document.getElementById('fsm-form').addEventListener('submit', async (e) => {
        e.preventDefault();
        const log = document.getElementById('sim-log');
        const concept = document.getElementById('sim-concept').value;
        const domain = document.getElementById('sim-domain').value;
        
        // Clear log
        log.innerHTML = '';
        
        const addLog = (msg, type='info') => {
            const div = document.createElement('div');
            div.className = `log-line ${type}`;
            div.innerHTML = msg;
            log.appendChild(div);
            log.scrollTop = log.scrollHeight;
        };

        const sleep = ms => new Promise(r => setTimeout(r, ms));

        addLog(`> Initializing FSM Task for: "${concept}"`, 'system');
        await sleep(600);

        addLog(`> Context: ${domain}`, 'system');
        addLog(`> Step 1: Identifying States...`, 'info');
