    // Main Tab Switcher
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.style.display = 'none');
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).style.display = 'block';
        // Find the button that called this and set active
        const buttons = document.querySelectorAll('.tabs-nav button');
        if (tabId === 'docs') buttons[0].classList.add('active');
        if (tabId === 'preview') buttons[1].classList.add('active');
        if (tabId === 'integration') buttons[2].classList.add('active');
    }

    // --- Simulation Logic ---
    document.addEventListener('DOMContentLoaded', () => {
        const state = { isRunning: false, simulationSpeed: 1 };
        
        const ui = {
            runBtn: document.getElementById('btnRunSim'),
            simTabs: document.querySelectorAll('.sim-tab'),
            panels: {
                viz: document.getElementById('sim-viz-panel'),
                log: document.getElementById('sim-log-panel'),
                code: document.getElementById('sim-code-panel')
            },
            inputs: {
                depth: document.getElementById('simMaxDepth'),
                candidates: document.getElementById('simCandidateRules')
            },
            terminal: document.getElementById('terminalOutput'),
            svg: document.getElementById('treeSvg'),
            codeBlock: document.querySelector('#codeOutput code')
        };

        const wait = (ms) => new Promise(resolve => setTimeout(resolve, ms * state.simulationSpeed));

        // Simulation Tab Switcher
        ui.simTabs.forEach(tab => {
            tab.addEventListener('click', () => {
                ui.simTabs.forEach(t => t.classList.remove('active'));
                tab.classList.add('active');
                
                const target = tab.dataset.target;
                Object.values(ui.panels).forEach(p => p.style.display = 'none');
                ui.panels[target].style.display = 'block';
            });
        });

        const logger = {
            clear: () => ui.terminal.innerHTML = '',
            log: (msg, type = 'info') => {
                const div = document.createElement('div');
                div.className = `log-entry log-${type}`;
                const time = new Date().toLocaleTimeString('en-US', { hour12: false, hour: "numeric", minute: "numeric", second: "numeric" });
                div.innerHTML = `<span style="opacity:0.5">[${time}]</span> > ${msg}`;
                ui.terminal.appendChild(div);
                ui.terminal.scrollTop = ui.terminal.scrollHeight;
            }
        };

        const visualizer = {
            clear: () => ui.svg.innerHTML = '',
            drawNode: (x, y, label, color = '#1e293b') => {
                const g = document.createElementNS("http://www.w3.org/2000/svg", "g");
                g.setAttribute("class", "node");
                
                const circle = document.createElementNS("http://www.w3.org/2000/svg", "circle");
                circle.setAttribute("cx", x); circle.setAttribute("cy", y); circle.setAttribute("r", 20);
                circle.style.fill = color;
                
                const text = document.createElementNS("http://www.w3.org/2000/svg", "text");
                text.setAttribute("x", x); text.setAttribute("y", y);
                text.textContent = label;
                
                g.appendChild(circle); g.appendChild(text);
                ui.svg.appendChild(g);
            },
            drawLink: (x1, y1, x2, y2, label) => {
                const line = document.createElementNS("http://www.w3.org/2000/svg", "line");
                line.setAttribute("x1", x1); line.setAttribute("y1", y1);
                line.setAttribute("x2", x2); line.setAttribute("y2", y2);
                line.setAttribute("class", "link");
                ui.svg.prepend(line);

                if(label) {
                    const text = document.createElementNS("http://www.w3.org/2000/svg", "text");
                    text.setAttribute("x", (x1+x2)/2); text.setAttribute("y", (y1+y2)/2 - 5);
                    text.setAttribute("class", "node-label");
                    text.setAttribute("text-anchor", "middle");
                    text.textContent = label;
                    ui.svg.appendChild(text);
                }
            },
            renderStep: (step) => {
                visualizer.clear();
                const width = ui.svg.clientWidth || 600;
                const startX = width / 2;
                const startY = 40;
                const levelH = 80;
                const spread = 120;

                visualizer.drawNode(startX, startY, "Root");

                if (step >= 1) {
                    visualizer.drawLink(startX, startY, startX - spread, startY + levelH, "Contract=M2M");
                    visualizer.drawLink(startX, startY, startX + spread, startY + levelH, "Else");
                    visualizer.drawNode(startX - spread, startY + levelH, step === 1 ? "?" : "Split");
                    visualizer.drawNode(startX + spread, startY + levelH, step === 1 ? "?" : "No", step >= 3 ? "#4ade80" : "#1e293b");
                }
                if (step >= 2) {
                    const leftX = startX - spread;
                    const leftY = startY + levelH;
                    visualizer.drawLink(leftX, leftY, leftX - 60, leftY + levelH, "Usage<10");
                    visualizer.drawLink(leftX, leftY, leftX + 60, leftY + levelH, "Else");
                    visualizer.drawNode(leftX - 60, leftY + levelH, "Yes", "#f87171");
                    visualizer.drawNode(leftX + 60, leftY + levelH, "No", "#4ade80");
                }
            }
        };

        const runSimulation = async () => {
            if (state.isRunning) return;
            state.isRunning = true;
            ui.runBtn.textContent = "Running...";
            ui.runBtn.style.opacity = "0.7";

            try {
                // Switch to Log tab
                ui.simTabs[1].click();
                logger.clear();
                logger.log(`Initializing DecisionTreeTask...`);
                logger.log(`Config: Target='Churn', Depth=${ui.inputs.depth.value}`);
                await wait(800);
                
                logger.log(`Loading data... 1000 records.`);
                logger.log(`Initial Entropy: 0.98`);
                await wait(800);

                logger.log(`Depth 0: Analyzing root...`);
                await wait(600);
                logger.log(`LLM Candidates: Contract, Age, Usage`, 'info');
                await wait(600);
                logger.log(`Best Split: Contract == 'Month-to-Month' (Gain: 0.45)`, 'success');

                ui.simTabs[0].click(); // Viz
                visualizer.renderStep(1);
                await wait(1500);

                ui.simTabs[1].click(); // Log
                logger.log(`Depth 1 (Left): Subset size 450.`);
                logger.log(`LLM Candidates: LastCall, Usage`, 'info');
                await wait(800);
                logger.log(`Best Split: Usage < 10.5 (Gain: 0.22)`, 'success');

                ui.simTabs[0].click(); // Viz
                visualizer.renderStep(2);
                await wait(1500);

                ui.simTabs[1].click(); // Log
                logger.log(`Tree construction complete.`);
                logger.log(`Generating Kotlin code...`);
                await wait(800);

                const code = `fun predict(record: Map<String, String>): String {
    if (record["Contract"] == "Month-to-Month") {
        val usage = record["Usage"]?.toDoubleOrNull() ?: 0.0
        if (usage < 10.5) return "Yes"
        else return "No"
    } else {
        return "No"
    }
}`;
                ui.codeBlock.textContent = code;
                ui.simTabs[2].click(); // Code

            } catch(e) {
                logger.log(`Error: ${e}`, 'error');
            } finally {
                state.isRunning = false;
                ui.runBtn.textContent = "Run Task";
                ui.runBtn.style.opacity = "1";
            }
        };

        ui.runBtn.addEventListener('click', runSimulation);
        visualizer.renderStep(0);
    });
