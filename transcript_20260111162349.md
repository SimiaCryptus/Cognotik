    // Tab Switching Logic
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.style.display = 'none');
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).style.display = 'block';
        event.target.classList.add('active');
    }

    // Simulator Logic
    function loadPreset(type) {
        if (type === 'calculus') {
            document.getElementById('sim-problem').value = "Calculate the integral of x^2 dx";
            document.getElementById('sim-domain').value = "calculus";
            document.getElementById('sim-goal').value = "Find indefinite integral";
        } else if (type === 'algebra') {
            document.getElementById('sim-problem').value = "Solve for x: 2x + 5 = 15";
            document.getElementById('sim-domain').value = "algebra";
            document.getElementById('sim-goal').value = "Isolate x";
        }
    }

    function runSimulation() {
        const container = document.getElementById('sim-results');
        const domain = document.getElementById('sim-domain').value;
        
        container.innerHTML = '<div style="color: var(--accent-primary);">Analyzing problem structure...</div>';

        setTimeout(() => {
            let steps = [];
            
            if (domain === 'calculus') {
                steps = [
                    { title: 'Identify Rule', latex: '\\int x^n dx = \\frac{x^{n+1}}{n+1} + C', just: 'Power Rule for Integration' },
                    { title: 'Substitution', latex: '\\text{Set } n = 2 \\implies \\frac{x^{2+1}}{2+1} + C', just: 'Variable substitution' },
                    { title: 'Simplification', latex: '\\frac{x^3}{3} + C', just: 'Arithmetic' }
                ];
            } else if (domain === 'algebra') {
                steps = [
                    { title: 'Subtraction Property', latex: '2x = 15 - 5 \\implies 2x = 10', just: 'Subtract 5 from both sides' },
                    { title: 'Division Property', latex: 'x = \\frac{10}{2}', just: 'Divide by coefficient' },
                    { title: 'Solution', latex: 'x = 5', just: 'Simplification' }
                ];
            } else {
                steps = [
                    { title: 'Assumption', latex: '\\neg P \\implies Q', just: 'Hypothesis generation' },
                    { title: 'Inference', latex: 'Q \\land R \\implies \\text{False}', just: 'Logical deduction' },
                    { title: 'Conclusion', latex: '\\therefore P \\text{ is true}', just: 'Reductio ad absurdum' }
                ];
            }

            let html = `
                <div style="margin-bottom: 1rem; padding-bottom: 0.5rem; border-bottom: 1px solid var(--border-subtle);">
                    <strong>Status:</strong> <span style="color: #4ade80">Solved</span> &bull; 
                    <strong>Confidence:</strong> 99.8%
                </div>
            `;

            steps.forEach((step, index) => {
                html += `
                    <div class="math-step">
                        <h4>Step ${index + 1}: ${step.title}</h4>
                        <div>$$${step.latex}$$</div>
                        <div class="justification">${step.just}</div>
                    </div>
                `;
            });

            container.innerHTML = html;
            
            // Trigger MathJax re-render
            if (window.MathJax) {
                MathJax.typesetPromise([container]);
            }
        }, 800);
