    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.style.display = 'none');
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).style.display = 'block';
        event.target.classList.add('active');
    }

    function runSimulation() {
        const output = document.getElementById('sim-output');
        const goal = document.getElementById('sim-goal').value;
        const audience = document.getElementById('sim-audience').value;
        const type = document.getElementById('sim-type').value;

        output.innerHTML = '';
        
        const logs = [
            `[INFO] Initializing EmailCampaignTask...`,
            `[INFO] Goal: ${goal}`,
            `[INFO] Phase 1: Developing strategy for ${type} campaign...`,
            `[INFO] Analyzing audience: ${audience}...`,
            `[INFO] Phase 2: Outlining sequence (3 emails)...`,
            `[INFO] Phase 3: Generating content...`,
            `[INFO] Creating A/B subject line variants...`,
            `[SUCCESS] Campaign generation complete.`
        ];

        let i = 0;
        function printLog() {
            if (i < logs.length) {
                const div = document.createElement('div');
                div.textContent = `> ${logs[i]}`;
                div.style.color = logs[i].includes('SUCCESS') ? '#4ade80' : '#8b949e';
                div.style.marginBottom = '4px';
                output.appendChild(div);
                output.scrollTop = output.scrollHeight;
                i++;
                setTimeout(printLog, 300);
            } else {
                renderMockResult(goal, audience);
            }
        }
        printLog();
    }
