    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.style.display = 'none');
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));
        document.getElementById('tab-' + tabId).style.display = 'block';
        event.target.classList.add('active');
    }

    function loadPreset(type) {
        if (type === 'remote') {
            document.getElementById('sim-thesis').value = "Remote work increases productivity and employee satisfaction.";
            document.getElementById('sim-antithesis').value = "In-office work fosters collaboration and company culture.";
        } else if (type === 'arch') {
            document.getElementById('sim-thesis').value = "Move fast and break things to capture market share.";
            document.getElementById('sim-antithesis').value = "Maintain strict quality controls and zero-bug policy to build trust.";
        }
    }

    function runSimulation() {
        const thesis = document.getElementById('sim-thesis').value;
        const antithesis = document.getElementById('sim-antithesis').value;
        const logsEl = document.getElementById('sim-output');
        
        if(!thesis || !antithesis) {
            logsEl.innerHTML = '<div class="log-entry" style="color:red">Error: Both Thesis and Antithesis are required.</div>';
            return;
        }

        logsEl.innerHTML = '';
        
        const steps = [
            { msg: `[INFO] Analyzing Thesis: "${thesis.substring(0, 30)}..."`, delay: 500 },
            { msg: `[INFO] Analyzing Antithesis: "${antithesis.substring(0, 30)}..."`, delay: 1500 },
            { msg: `[INFO] Exploring contradictions and tensions...`, delay: 2500 },
            { msg: `[INFO] Generating Synthesis Level 1...`, delay: 3500 },
            { msg: `[INFO] Generating Synthesis Level 2...`, delay: 4500 },
            { msg: `[SUCCESS] Final Integration Complete.`, delay: 5500, type: 'success' }
        ];
        
        let startTime = Date.now();

        steps.forEach(step => {
            setTimeout(() => {
                const line = document.createElement('div');
                line.className = 'log-entry ' + (step.type === 'success' ? 'log-success' : 'log-info');
                line.textContent = `> ${new Date().toLocaleTimeString()} ${step.msg}`;
                logsEl.appendChild(line);
                logsEl.scrollTop = logsEl.scrollHeight;
            }, step.delay);
        });

        setTimeout(() => {
            const result = document.createElement('div');
            result.className = 'log-result';
            result.innerHTML = `
                <strong>Synthesis Result:</strong><br><br>
                A hybrid model that mandates specific "collaboration days" for in-office presence while allowing flexible remote work for deep focus tasks effectively bridges the gap. This approach preserves the autonomy and productivity of remote work while ensuring the social cohesion and spontaneous innovation of physical presence.
            `;
            logsEl.appendChild(result);

        }, 6000);
    }
