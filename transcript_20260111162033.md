    // Main Tab Switcher
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => {
            el.style.display = 'none';
            el.classList.remove('active');
        });
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        const target = document.getElementById('tab-' + tabId);
        if(target) {
            target.style.display = 'block';
            setTimeout(() => target.classList.add('active'), 10);
        }
        event.target.classList.add('active');
    }

    // Sample Sub-Tab Switcher
    function switchSample(sampleId, btn) {
        document.querySelectorAll('.sample-pane').forEach(el => el.style.display = 'none');
        document.getElementById('sample-' + sampleId).style.display = 'block';
        
        // Update buttons in the sample nav
        btn.parentElement.querySelectorAll('button').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
    }

    // Calculator Logic
    document.addEventListener('DOMContentLoaded', () => {
        const elements = {
            slider: document.getElementById('wordCountSlider'),
            display: document.getElementById('wordCountDisplay'),
            revSlider: document.getElementById('revisionSlider'),
            revDisplay: document.getElementById('revisionDisplay'),
            acts: document.getElementById('actsSelect'),
            scenes: document.getElementById('scenesSelect'),
            outputs: {
                totalScenes: document.getElementById('totalScenes'),
                wordsPerScene: document.getElementById('wordsPerScene'),
                images: document.getElementById('imageCount'),
                time: document.getElementById('estTime')
            }
        };

        function updateStats() {
            if (!elements.slider) return;

            const totalWords = parseInt(elements.slider.value);
            const revisions = parseInt(elements.revSlider.value);
            const numActs = parseInt(elements.acts.value);
            const scenesPerAct = parseInt(elements.scenes.value);

            elements.display.textContent = totalWords.toLocaleString();
            elements.revDisplay.textContent = revisions;

            const totalScenes = numActs * scenesPerAct;
            const wordsPerScene = totalScenes > 0 ? Math.round(totalWords / totalScenes) : 0;
            const imageCount = 1 + totalScenes + 6; // Cover + Scenes + Chars/Settings
            const timePerScene = 0.5 + (revisions * 0.2);
            const timeEstimate = (totalScenes * timePerScene).toFixed(1);

            elements.outputs.totalScenes.textContent = totalScenes;
            elements.outputs.wordsPerScene.textContent = wordsPerScene;
            elements.outputs.images.textContent = imageCount;
            elements.outputs.time.textContent = `${timeEstimate} min`;
        }

        if (elements.slider) {
            elements.slider.addEventListener('input', updateStats);
            elements.revSlider.addEventListener('input', updateStats);
            elements.acts.addEventListener('change', updateStats);
            elements.scenes.addEventListener('change', updateStats);
            updateStats();
        }
    });
