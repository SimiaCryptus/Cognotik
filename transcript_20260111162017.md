    // Main Page Tab Switcher
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.classList.remove('active'));
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).classList.add('active');
        event.target.classList.add('active');
    }

    // Simulator Tab Switcher
    window.switchSimTab = function(tabId) {
        document.querySelectorAll('.sim-content').forEach(el => el.classList.remove('active'));
        document.querySelectorAll('.sim-tab-btn').forEach(el => el.classList.remove('active'));

        document.getElementById('sim-' + tabId).classList.add('active');
        // Find button that triggered this (approximate)
        const btns = document.querySelectorAll('.sim-tab-btn');
        btns.forEach(btn => {
            if(btn.getAttribute('onclick').includes(tabId)) btn.classList.add('active');
        });
    }

    // --- SIMULATION LOGIC ---
    document.addEventListener('DOMContentLoaded', () => {
        const rangeInput = document.getElementById('target_pages');
        const rangeVal = document.getElementById('page-val');
        if(rangeInput) {
            rangeInput.addEventListener('input', (e) => rangeVal.textContent = e.target.value);
        }

        const form = document.getElementById('task-config-form');
        if(form) {
            form.addEventListener('submit', (e) => {
                e.preventDefault();
                runSimulation();
            });
        }
    });

    const delay = (ms) => new Promise(resolve => setTimeout(resolve, ms));

    async function runSimulation() {
        const subject = document.getElementById('subject').value || "Generic Hero Story";
        const style = document.getElementById('art_style').value;
        const pages = document.getElementById('target_pages').value;
        const genImages = document.getElementById('generate_images').checked;
        const runBtn = document.getElementById('run-btn');
        const logDisplay = document.getElementById('log-display');

        // Reset
        runBtn.disabled = true;
        runBtn.innerText = "Processing...";
        logDisplay.innerHTML = '<div style="color: #64748b;">// Initializing ComicBookGenerationTask...</div>';
        document.getElementById('script-display').innerHTML = '';
        document.getElementById('char-display').innerHTML = '';
        document.getElementById('visuals-display').innerHTML = '';
        
        switchSimTab('logs');

        const log = (msg, color='#fff') => {
            logDisplay.innerHTML += `<div style="margin-top:4px; color:${color}">> ${msg}</div>`;
            logDisplay.scrollTop = logDisplay.scrollHeight;
        };

        try {
            await delay(800);
            log(`Analyzing subject: "${subject}"...`);
            log(`Style set to: ${style}`);

            // Step 1: Script
            await delay(1200);
            log("Generating script structure...", "#fbbf24");
            
            const scriptHtml = `
                <h3 style="color:var(--accent-primary)">${subject.substring(0,20).toUpperCase()}...</h3>
                <p><strong>Page 1</strong></p>
                <div style="margin-left:1rem; border-left:2px solid #333; padding-left:1rem;">
                    <p><strong>Row 1</strong></p>
                    <p>Panel 1: Establishing shot. <br><span style="color:#888">CAPTION: "It began on a Tuesday."</span></p>
                    <p>Panel 2: Close up on protagonist. <br><span style="color:#888">HERO: "Wait, what is that?"</span></p>
                </div>
            `;
            document.getElementById('script-display').innerHTML = scriptHtml;
            log("Script generated successfully.", "#4ade80");

            // Step 2: Characters
            await delay(1000);
            log("Extracting character profiles...", "#fbbf24");
            const chars = [
                {name: "Protagonist", desc: "Determined, weary"},
                {name: "Antagonist", desc: "Shadowy, mysterious"}
            ];
            chars.forEach(c => {
                document.getElementById('char-display').innerHTML += `
                    <div class="char-card">
                        <div class="char-img-placeholder">[IMG: ${c.name}]</div>
                        <strong>${c.name}</strong><br>
                        <small>${c.desc}</small>
                    </div>
                `;
            });
            log(`Defined ${chars.length} characters.`, "#4ade80");

            // Step 3: Visuals
            if(genImages) {
                await delay(1000);
                log("Rendering visual panels (GPU)...", "#fbbf24");
                await delay(1500);
                document.getElementById('visuals-display').innerHTML = `
                    <div style="background:#000; padding:1rem; border:1px solid #333; text-align:center; border-radius:4px;">
                        <div style="height:150px; background:linear-gradient(45deg, #111, #222); display:flex; align-items:center; justify-content:center; color:#555;">
                            [Generated Comic Strip: Page 1 Row 1]
                        </div>
                        <small style="color:#666; display:block; margin-top:0.5rem;">Style: ${style}</small>
                    </div>
                `;
                log("Visuals rendered.", "#4ade80");
                switchSimTab('visuals');
            } else {
                switchSimTab('script');
            }

            log("Task Complete.", "#4ade80");

        } catch (e) {
            log("Error: " + e.message, "#ef4444");
        } finally {
            runBtn.disabled = false;
            runBtn.innerText = "Run Task";
        }
    }
