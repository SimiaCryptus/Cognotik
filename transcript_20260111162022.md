    function runSimulation() {
        const topic = document.getElementById('sim_topic').value;
        const format = document.getElementById('sim_format').value;
        const style = document.getElementById('sim_style').value;
        const pub = document.getElementById('sim_pub').value;
        const outputEl = document.getElementById('sim-output');

        outputEl.innerHTML = ''; // Clear previous output

        const logs = [
            `[INFO] Initializing ArticleGenerationTask...`,
            `[INFO] Topic: "${topic}"`,
            `[INFO] Config: ${format}, ${style}, Target: ${pub}`,
            `[INFO] Phase 1: Running journalism investigation...`,
            `[INFO] Analyzing context and verifying facts...`,
            `[INFO] Phase 2: Generating article structure...`,
            `[INFO] Creating outline: Headline, Lede, Sections, Conclusion...`,
            `[INFO] Phase 3: Writing article draft...`,
            `[INFO] Applying ${pub} tone and ${style}...`,
            `[SUCCESS] Article generation complete.`
        ];

        let logIndex = 0;
        
        // Create a container for logs
        const logContainer = document.createElement('div');
        logContainer.style.fontFamily = 'var(--font-mono)';
        logContainer.style.marginBottom = '2rem';
        outputEl.appendChild(logContainer);

        const logInterval = setInterval(() => {
            if (logIndex < logs.length) {
                const line = document.createElement('div');
                line.className = 'log-line';
                line.textContent = `> ${new Date().toLocaleTimeString()} ${logs[logIndex]}`;
                
                if (logs[logIndex].includes('SUCCESS')) line.classList.add('log-success');
                else if (logs[logIndex].includes('INFO')) line.classList.add('log-info');
                
                logContainer.appendChild(line);
                logIndex++;
            } else {
                clearInterval(logInterval);
                renderFinalArticle(outputEl, topic, pub);
            }
        }, 300);
    }
