    const mockFileContent = [
        "package com.simiacryptus.cognotik.plan",
        "",
        "import com.simiacryptus.cognotik.util.ValidatedObject",
        "",
        "abstract class TaskExecutionConfig(",
        "    val task_type: String,",
        "    val task_description: String? = null,",
        "    val task_dependencies: MutableList<String>? = null,",
        "    val state: TaskState? = null",
        ") : ValidatedObject {",
        "    // Base configuration logic",
        "    open fun validate(): String? = null",
        "}",
        "",
        "class SearchTaskExecutionConfigData(",
        "    val search_pattern: String = \"\",",
        "    val is_regex: Boolean = false",
        ") : TaskExecutionConfig(",
        "    task_type = \"FileSearch\",",
        "    task_description = \"Search files\"",
        ") {",
        "    // Implementation details",
        "}"
    ];

    const els = {
        pattern: document.getElementById('simPattern'),
        regex: document.getElementById('simRegex'),
        context: document.getElementById('simContext'),
        output: document.getElementById('simOutput')
    };

    function updateSimulation() {
        const pattern = els.pattern.value;
        const context = parseInt(els.context.value);
        const isRegex = els.regex.checked;

        if (!pattern) {
            els.output.innerHTML = '<div style="color:#64748b">Enter a pattern to search...</div>';
            return;
        }

        let matches = [];
        try {
            const regex = isRegex ? new RegExp(pattern) : new RegExp(pattern.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'));
            mockFileContent.forEach((line, index) => {
                if (regex.test(line)) matches.push(index);
            });
        } catch (e) {
            els.output.innerHTML = `<div style="color: #ef4444">Invalid Regex Pattern</div>`;
            return;
        }

        if (matches.length === 0) {
            els.output.innerHTML = '<div style="color:#64748b">No matches found.</div>';
            return;
        }

        // Group matches
        let blocks = [];
        let currentBlock = null;
        matches.forEach(lineIdx => {
            const start = Math.max(0, lineIdx - context);
            const end = Math.min(mockFileContent.length - 1, lineIdx + context);
            
            if (currentBlock && start <= currentBlock.end + 1) {
                currentBlock.end = Math.max(currentBlock.end, end);
                currentBlock.matches.push(lineIdx);
            } else {
                if (currentBlock) blocks.push(currentBlock);
                currentBlock = { start, end, matches: [lineIdx] };
            }
        });
        if (currentBlock) blocks.push(currentBlock);

        // Render
        let html = `<div style="margin-bottom:1rem; color:#94a3b8">Found ${matches.length} match(es) in mock file.</div>`;
        html += `<div style="color: #e2e8f0; font-weight: bold; margin-bottom: 0.5rem;">## src/main/kotlin/TaskConfig.kt</div>`;
        
        blocks.forEach(block => {
            html += `<div style="margin-top: 1rem; border-left: 2px solid #334155; padding-left: 1rem;">`;
            html += `<div style="color: #475569; font-size: 0.75rem; margin-bottom: 0.5rem;">Lines ${block.start + 1} - ${block.end + 1}</div>`;
            
            for (let i = block.start; i <= block.end; i++) {
                const isMatch = block.matches.includes(i);
                const lineClass = isMatch ? "line-match" : "line-context";
                const prefix = isMatch ? ">" : " ";
                const content = mockFileContent[i].replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
                
                html += `<div class="${lineClass}">`;
                html += `<span class="line-num">${i + 1}</span>`;
                html += `<span style="user-select:none; margin-right:5px;">${prefix}</span>`;
                html += `${content}</div>`;
            }
            html += `</div>`;
        });

        els.output.innerHTML = html;
    }

    [els.pattern, els.regex, els.context].forEach(el => {
        el.addEventListener('input', updateSimulation);
        el.addEventListener('change', updateSimulation);
    });
