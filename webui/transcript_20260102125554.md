package com.simiacryptus.cognotik.plan.tools.writing

import com.simiacryptus.cognotik.plan.model.Task
import com.simiacryptus.cognotik.plan.model.TaskContext
import com.simiacryptus.cognotik.plan.model.TaskResult
import org.slf4j.LoggerFactory

class ProbabilisticReasoningTask(
    val context: String,
    val hypotheses: List<String>,
    val evidence: List<String>,
    val riskTolerance: String = "medium"
) : Task() {

    override fun run(context: TaskContext): TaskResult {
        val prompt = """
            # Probabilistic Reasoning Analysis
            
            **Context:** ${this.context}
            **Risk Tolerance:** ${this.riskTolerance}
            
            **Hypotheses:**
            ${hypotheses.joinToString("\n") { "* $it" }}
            
            **Evidence:**
            ${evidence.joinToString("\n") { "* $it" }}
            
            Perform a comprehensive Bayesian analysis and decision support assessment based on the provided context, hypotheses, and evidence.
            
            Structure the response as follows:
            
            ## Bayesian Update
            - **Likelihood Assessment:** Create a table of $P(E|H)$ for each evidence and hypothesis. Include a "Diagnostic Value" column explaining the discriminatory power of the evidence.
            - **Posterior Probabilities:** Calculate $P(H|E)$. Show the table with Prior, Likelihood, and Posterior probabilities.
            - **Probability Changes:** Analyze the shift from Prior to Posterior for each hypothesis.
            - **Confidence Analysis:** Assess the confidence level, assumptions, and any ambiguity in the conclusion.
            - **Key Insights:** Summarize the most important findings from the update.
            
            ## Expected Value Analysis
            - **Expected Value (EV) Summary:** Define a utility function (Success/Failure/Cost) and calculate EV for potential actions corresponding to each hypothesis. Present this in a table.
            - **Risk Assessment:** Analyze worst-case scenarios, variance, and downside risk.
            - **Decision Recommendation:** Recommend a course of action based on the Risk Tolerance (${this.riskTolerance}).
            - **Sensitivity Analysis:** How sensitive is the decision to key variables or evidence reliability?
            - **Value of Information (EVPI):** What is the value of obtaining perfect information?
            
            ## Key Uncertainties
            - **Top Uncertainties:** Identify critical unknowns that could flip the diagnosis.
            - **Impact Assessment:** Perform a sensitivity analysis on these uncertainties.
            - **Confidence Intervals:** Estimate credible intervals for the hypotheses.
            - **Information Priorities:** What information is most needed to resolve the remaining uncertainty?
            
            ## Suggested Experiments
            - **Recommended Experiments:** Propose specific, actionable experiments to reduce uncertainty.
            - **Expected Information Gain (EIG):** Estimate the impact of these experiments on the posterior probabilities.
            - **Sequential Testing Strategy:** Recommend an order of operations to minimize Mean Time to Resolution (MTTR).
            - **Practical Considerations:** Resources, risks, and constraints.
            - **Decision Criteria:** Define clear "Stop" conditions for when to act.
            
            Use Markdown for formatting, including tables and LaTeX for math (e.g., $P(E|H)$). Ensure the tone is analytical and professional.
        """.trimIndent()

        val result = context.llm.chat(prompt)
        return TaskResult(result)
    }

    companion object {
        private val log = LoggerFactory.getLogger(ProbabilisticReasoningTask::class.java)
    }
}
