    // Main Tab Switcher
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => {
            el.classList.remove('active');
            el.style.display = 'none'; // Force hide for animation reset
        });
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        const target = document.getElementById('tab-' + tabId);
        target.style.display = 'block';
        // Small timeout to allow display:block to apply before adding active class for animation
        setTimeout(() => target.classList.add('active'), 10);
        
        event.target.classList.add('active');
    }

    // Simulator View Switcher
    function switchSimView(viewId) {
        document.querySelectorAll('.output-view').forEach(el => el.classList.remove('active'));
        document.querySelectorAll('.output-tab').forEach(el => el.classList.remove('active'));

        document.getElementById('view-' + viewId).classList.add('active');
        event.target.classList.add('active');
    }

    // Simulator Logic
    document.getElementById('simForm').addEventListener('submit', function(e) {
        e.preventDefault();
        
        const name = document.getElementById('simName').value;
        const desc = document.getElementById('simDesc').value;
        const stack = document.getElementById('simStack').value.split(',').map(s => s.trim());
        
        // Switch to logs
        const logTab = document.querySelector('.output-tab:nth-child(3)');
        logTab.click();
        
        const logContainer = document.getElementById('view-logs');
        logContainer.innerHTML = '';

        const addLog = (msg, type='info') => {
            const time = new Date().toLocaleTimeString().split(' ')[0];
            const colorClass = type === 'success' ? 'log-success' : 'log-info';
            logContainer.innerHTML += `<div class="log-line"><span class="log-time">${time}</span> <span class="${colorClass}">[${type.toUpperCase()}]</span> ${msg}</div>`;
            logContainer.scrollTop = logContainer.scrollHeight;
        };

        addLog(`Initializing task for '${name}'...`);
        
        setTimeout(() => addLog(`Analyzing description: "${desc.substring(0, 20)}..."`), 600);
        setTimeout(() => addLog(`Detected Tech Stack: ${stack.join(', ')}`), 1200);
        setTimeout(() => addLog(`Generating C4 Context Diagram...`), 1800);
        setTimeout(() => addLog(`Structuring Agile Epics and Sprints...`), 2400);
        
        setTimeout(() => {
            addLog(`Generation Complete.`, 'success');
            generateArtifacts(name, stack);
            // Switch back to diagram
            document.querySelector('.output-tab:nth-child(1)').click();
        }, 3000);
    });

    function generateArtifacts(name, stack) {
        // 1. Generate Mermaid
        const techNodes = stack.slice(0, 3).map((t, i) => `    Svc${i}[${t} Service]`).join('\n');
        const mermaid = `graph TB
    User[End User] --> LB[Load Balancer]
    LB --> API[API Gateway]
${techNodes}
    API --> Svc0
    Svc0 --> DB[(Database)]
    classDef primary fill:#1e293b,stroke:#38bdf8,stroke-width:2px;
    class User,LB,API,Svc0,DB primary;`;
        
        document.getElementById('mermaid-output').textContent = mermaid;

        // 2. Generate JSON
        const json = {
            project: name,
            generated_at: new Date().toISOString(),
            epics: [
                { id: "EPIC-1", name: "Infrastructure Setup", points: 13 },
                { id: "EPIC-2", name: "Core API Implementation", points: 21 }
            ],
            sprints: [
                { id: "SP-1", goal: "Hello World & DB Conn", tasks: ["Setup Repo", "Docker Compose"] },
                { id: "SP-2", goal: "MVP Features", tasks: ["User Auth", "Basic CRUD"] }
            ]
        };
        document.getElementById('json-output').textContent = JSON.stringify(json, null, 2);
    }
