    // --- Simulator Logic ---
    
    let hypotheses = [
        { name: "Database Lock Contention", prob: 0.4 },
        { name: "Network Latency", prob: 0.3 },
        { name: "Memory Leak", prob: 0.2 },
        { name: "Disk I/O Saturation", prob: 0.1 }
    ];

    function renderHypotheses() {
        const container = document.getElementById('hypotheses-list');
        container.innerHTML = '';
        hypotheses.forEach((h, index) => {
            const row = document.createElement('div');
            row.className = 'hypothesis-row';
            row.innerHTML = `
                <input type="text" value="${h.name}" onchange="updateHypothesis(${index}, 'name', this.value)">
                <input type="number" step="0.01" value="${h.prob}" onchange="updateHypothesis(${index}, 'prob', this.value)">
                <button class="btn-icon" onclick="removeHypothesis(${index})">×</button>
            `;
            container.appendChild(row);
        });
    }

    function updateHypothesis(index, field, value) {
        if (field === 'prob') value = parseFloat(value);
        hypotheses[index][field] = value;
    }

    function addHypothesis() {
        hypotheses.push({ name: "New Hypothesis", prob: 0.0 });
        renderHypotheses();
    }

    function removeHypothesis(index) {
        hypotheses.splice(index, 1);
        renderHypotheses();
    }

    function loadPreset(type) {
        if (type === 'root_cause') {
            document.getElementById('sim-context').value = "System is experiencing intermittent 500 errors during peak load.";
            document.getElementById('sim-evidence').value = "Database CPU is at 40%\nWeb server logs show connection timeouts";
            hypotheses = [
                { name: "Database Lock Contention", prob: 0.4 },
                { name: "Network Latency", prob: 0.3 },
                { name: "Memory Leak", prob: 0.2 },
                { name: "Disk I/O Saturation", prob: 0.1 }
            ];
        } else if (type === 'market') {
            document.getElementById('sim-context').value = "Competitor launched a similar feature. Should we expedite launch?";
            document.getElementById('sim-evidence').value = "Competitor pricing is 20% lower\nEarly user reviews mention bugs";
            hypotheses = [
                { name: "Market Share Loss (High)", prob: 0.5 },
                { name: "Market Indifference", prob: 0.3 },
                { name: "Competitor Failure", prob: 0.2 }
            ];
        }
        renderHypotheses();
    }

    function runSimulation() {
        const output = document.getElementById('sim-output-area');
        output.innerHTML = '<div style="text-align:center; margin-top:4rem; color:var(--text-dim);">Running Bayesian Analysis...</div>';
        
        setTimeout(() => {
            const context = document.getElementById('sim-context').value;
            const evidence = document.getElementById('sim-evidence').value.split('\n').filter(e => e.trim());
            const risk = document.getElementById('sim-risk').value;
            
            // Mock Logic: Shuffle probabilities
            const results = hypotheses.map(h => {
                const change = (Math.random() - 0.5) * 0.3; // Random swing
                let newProb = Math.max(0, Math.min(1, h.prob + change));
                return { ...h, newProb };
            });
            
            // Normalize
            const total = results.reduce((sum, h) => sum + h.newProb, 0);
            results.forEach(h => h.newProb = h.newProb / total);
            results.sort((a, b) => b.newProb - a.newProb);

            let md = `<div class="md-h1">Analysis Results</div>`;
            
            md += `<div class="md-h2">Bayesian Update</div>`;
            md += `<p>Based on the evidence provided, here are the updated beliefs:</p><br>`;
            
            results.forEach(h => {
                const pct = (h.newProb * 100).toFixed(1);
                const delta = ((h.newProb - h.prob) * 100).toFixed(1);
                const sign = delta >= 0 ? '+' : '';
                const colorClass = delta >= 0 ? 'prob-change-pos' : 'prob-change-neg';
                md += `<div><span class="md-bold">${h.name}</span>: ${pct}% (<span class="${colorClass}">${sign}${delta}%</span>)</div>`;
            });

            md += `<br><div class="md-h2">Likelihood Assessment</div>`;
            md += `<p>The evidence strongly suggests <span class="md-bold">${results[0].name}</span> is the most likely cause.</p>`;

            md += `<br><div class="md-h2">Risk Analysis (${risk} tolerance)</div>`;
            if (risk === 'high') {
                md += `<p>Given high risk tolerance, immediate action on <span class="md-bold">${results[0].name}</span> is recommended despite residual uncertainty.</p>`;
            } else {
                md += `<p>Given ${risk} risk tolerance, further validation is recommended before committing resources.</p>`;
            }

            md += `<br><div class="md-h2">Suggested Experiments</div>`;
            md += `<div class="md-list">1. Validate ${results[0].name} by isolating the subsystem.</div>`;
            md += `<div class="md-list">2. Collect more logs regarding ${evidence[0] ? evidence[0].substring(0, 15) + '...' : 'observed behavior'}.</div>`;

            output.innerHTML = md;
        }, 1000);
    }
