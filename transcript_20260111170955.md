<script>
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.classList.remove('active'));
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).classList.add('active');
        event.target.classList.add('active');
    }

    function clearSim() {
        document.getElementById('sim-output').innerHTML = '';
    }

    async function runSimulation() {
        const output = document.getElementById('sim-output');
        const type = document.getElementById('sim-log-type').value;
        output.innerHTML = '';

        const log = (msg, cls = 'log-info') => {
            const div = document.createElement('div');
            div.className = `log-line ${cls}`;
            div.innerHTML = `<span class="log-ts">${new Date().toLocaleTimeString()}</span> ${msg}`;
            output.appendChild(div);
            output.scrollTop = output.scrollHeight;
        };

        log("Initializing DataIngestTask...");
        await new Promise(r => setTimeout(r, 600));
        
        if (type === 'server') {
            log("Resolving files: **/*.log", "log-info");
            await new Promise(r => setTimeout(r, 400));
            log("Found 4 files. Sampling 1000 lines...", "log-info");
            await new Promise(r => setTimeout(r, 800));
            
            log("<strong>Phase 2: Pattern Discovery</strong>", "log-warn");
            await new Promise(r => setTimeout(r, 1000));
            
            // Iteration 1
            log("Iteration 1 | Coverage: 72% | Residuals: 280", "log-warn");
            log("✔ Discovered: `^(?<ts>\\S+) (?<level>\\w+) \\[(?<thread>.*)\\] (?<msg>.*)$`", "log-success");
            await new Promise(r => setTimeout(r, 1200));
            
            // Iteration 2
            log("Iteration 2 | Coverage: 96% | Residuals: 40", "log-warn");
            log("✔ Discovered: `^Caused by: (?<exception>[\\w\\.]+): (?<reason>.*)$`", "log-success");
            await new Promise(r => setTimeout(r, 500));
            
            log("Coverage threshold (0.95) met.", "log-success");
        } else {
            log("Resolving files: data/legacy/*.txt", "log-info");
            await new Promise(r => setTimeout(r, 400));
            log("Found 1 large file. Sampling...", "log-info");
            await new Promise(r => setTimeout(r, 800));
            
            log("<strong>Phase 2: Pattern Discovery</strong>", "log-warn");
            await new Promise(r => setTimeout(r, 1000));
            
            log("Iteration 1 | Coverage: 40% | Residuals: 600", "log-warn");
            log("✔ Discovered: `ID:(?<id>\\d+)\\|VAL:(?<val>\\d+\\.\\d+)`", "log-success");
            await new Promise(r => setTimeout(r, 1200));
            
            log("Iteration 2 | Coverage: 85% | Residuals: 150", "log-warn");
            log("✔ Discovered: `ERR:(?<code>\\d+) - (?<desc>.*)`", "log-success");
            await new Promise(r => setTimeout(r, 1200));

            log("Iteration 3 | Coverage: 99% | Residuals: 10", "log-warn");
            log("✔ Discovered: `FOOTER: (?<checksum>[A-F0-9]+)`", "log-success");
        }

        log("<strong>Phase 3: Bulk Extraction</strong>", "log-info");
        await new Promise(r => setTimeout(r, 600));
        log("Streaming data to data.jsonl...", "log-info");
        log("Building index.csv...", "log-info");
        await new Promise(r => setTimeout(r, 400));
        log("<strong>Ingestion Complete.</strong>", "log-success");
        
        // Show sample data
        const sampleJson = type === 'server' 
            ? `{"ts": "2023-11-01T10:00:00", "level": "INFO", "thread": "main", "msg": "System started"}`
            : `{"id": "4402", "val": "99.21"}`;
            
        const pre = document.createElement('pre');
        pre.style.color = 'var(--accent-secondary)';
        pre.style.marginTop = '1rem';
        pre.innerText = "Sample Record:\n" + JSON.stringify(JSON.parse(sampleJson), null, 2);
        output.appendChild(pre);
        output.scrollTop = output.scrollHeight;
    }
</script>
