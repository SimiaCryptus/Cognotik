    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.classList.remove('active'));
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).classList.add('active');
        event.target.classList.add('active');
    }

    function runSimulation() {
        const topic = document.getElementById('sim-topic').value;
        const anglesCount = document.getElementById('sim-angles').value;
        const outputContainer = document.getElementById('sim-output-container');
        
        outputContainer.innerHTML = '';
        
        const logs = [
            `[INFO] Initializing JournalismReasoningTask...`,
            `[INFO] Topic: ${topic}`,
            `[INFO] Verifying facts against knowledge base...`,
            `[INFO] Identifying source perspectives (Stakeholders: 4)...`,
            `[INFO] Analyzing historical context...`,
            `[INFO] Checking for framing biases...`,
            `[INFO] Exploring ${anglesCount} alternative angles...`,
            `[INFO] Identifying information gaps...`,
            `[INFO] Generating editorial synthesis...`,
            `[SUCCESS] Investigation complete.`
        ];
        
        let logIndex = 0;
        
        // Create a container for logs
        const logContainer = document.createElement('div');
        logContainer.style.fontFamily = 'monospace';
        logContainer.style.marginBottom = '1rem';
        logContainer.style.borderBottom = '1px solid #30363d';
        logContainer.style.paddingBottom = '1rem';
        outputContainer.appendChild(logContainer);

        const logInterval = setInterval(() => {
            if (logIndex < logs.length) {
                const line = document.createElement('div');
                line.textContent = `> ${new Date().toLocaleTimeString()} ${logs[logIndex]}`;
                line.style.marginBottom = '4px';
                
                if (logs[logIndex].includes('SUCCESS')) line.style.color = '#4ade80';
                else if (logs[logIndex].includes('INFO')) line.style.color = '#60a5fa';
                else line.style.color = '#8b949e';
                
                logContainer.appendChild(line);
                outputContainer.scrollTop = outputContainer.scrollHeight;
                logIndex++;
            } else {
                clearInterval(logInterval);
                renderReport(topic, anglesCount, outputContainer);
            }
        }, 300);
    }
