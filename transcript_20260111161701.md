<script>
    // ==========================================
    // 1. Main Tab Logic
    // ==========================================
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => {
            el.classList.remove('active');
            el.style.display = 'none';
        });
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        const target = document.getElementById('tab-' + tabId);
        target.style.display = 'block';
        // Small timeout to allow display:block to apply before adding active class for animation
        setTimeout(() => target.classList.add('active'), 10);
        
        // Find the button that triggered this (event might be null if called programmatically)
        if (event && event.target) {
            event.target.classList.add('active');
        }
    }

    // ==========================================
    // 2. Simulator Tab Logic
    // ==========================================
    function switchSimTab(tabId) {
        document.querySelectorAll('.sim-content').forEach(el => el.style.display = 'none');
        document.querySelectorAll('.sim-tab').forEach(el => el.classList.remove('active'));
        
        document.getElementById('sim-content-' + tabId).style.display = 'block';
        if (event && event.target) {
            event.target.classList.add('active');
        }
    }

    // ==========================================
    // 3. Simulator Execution Logic
    // ==========================================
    document.addEventListener('DOMContentLoaded', () => {
        const runBtn = document.getElementById('btn-sim-run');
        const codeDisplay = document.getElementById('display-code');
        const consoleDisplay = document.getElementById('display-console');
        const resultDisplay = document.getElementById('display-result');

        const log = (msg, type = 'info') => {
            const span = document.createElement('span');
            span.className = 'term-line';
            if (type === 'error') span.classList.add('text-error');
            if (type === 'success') span.classList.add('text-success');
            if (type === 'muted') span.classList.add('text-muted');
            span.innerText = `> ${msg}`;
            consoleDisplay.appendChild(span);
            consoleDisplay.scrollTop = consoleDisplay.scrollHeight;
        };

        const typeCode = async (text) => {
            codeDisplay.innerHTML = '';
            const lines = text.split('\n');
            for (let line of lines) {
                // Simple highlighting
                const html = line
                    .replace(/(def|return|if|for|import|new|class|val|var)/g, '<span style="color:#c678dd">$1</span>')
                    .replace(/(".+?")/g, '<span style="color:#98c379">$1</span>')
                    .replace(/(\/\/.*)/g, '<span style="color:#5c6370">$1</span>');
                
                codeDisplay.innerHTML += html + '\n';
                codeDisplay.parentElement.scrollTop = codeDisplay.parentElement.scrollHeight;
                await new Promise(r => setTimeout(r, 30));
            }
        };

        runBtn.addEventListener('click', async () => {
            const goal = document.getElementById('sim-goal').value;
            const workingDir = document.getElementById('sim-dir').value;
            const autoFix = document.getElementById('sim-autofix').checked;

            // Reset UI
            runBtn.disabled = true;
            runBtn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> Processing...';
            consoleDisplay.innerHTML = '';
            resultDisplay.innerText = '-- Pending --';

            // Step 1: Generate Code
            switchSimTab('code');
            log("Analyzing goal...", "muted");
            await new Promise(r => setTimeout(r, 800));
            
            let mockCode = `import java.io.File\n\n// Goal: ${goal}\n// Dir: ${workingDir}\n\n`;
            if (goal.toLowerCase().includes('fibonacci')) {
                mockCode += `def fib(n) {\n    if (n <= 1) return n\n    long a = 0, b = 1\n    (2..n).each { \n        long temp = a + b\n        a = b\n        b = temp\n    }\n    return b\n}\n\nprintln "Calculating..."\nreturn fib(100)`;
            } else {
                mockCode += `println "Executing generic task..."\nFile(workingDir).listFiles()?.each {\n    println "Found: \${it.name}"\n}\nreturn "Done"`;
            }
            
            await typeCode(mockCode);

            // Step 2: Execute
            await new Promise(r => setTimeout(r, 500));
            switchSimTab('console');
            log("Code generated. Requesting user approval...", "info");
            await new Promise(r => setTimeout(r, 800));
            log("User approved. Starting JVM...", "success");
            
            // Simulate Error/Autofix
            if (autoFix && Math.random() > 0.5) {
                await new Promise(r => setTimeout(r, 600));
                log("RuntimeException: NullPointerException at line 8", "error");
                log("Auto-Fix engaged. Analyzing stack trace...", "info");
                await new Promise(r => setTimeout(r, 1000));
                log("Patch applied. Retrying...", "success");
            }

            log("Execution finished. Exit code 0.", "success");

            // Step 3: Result
            await new Promise(r => setTimeout(r, 500));
            switchSimTab('result');
            if (goal.toLowerCase().includes('fibonacci')) {
                resultDisplay.innerText = "3736710778780434371";
            } else {
                resultDisplay.innerText = "Task completed successfully.";
            }
            resultDisplay.style.borderLeft = "4px solid #22c55e";

            runBtn.disabled = false;
            runBtn.innerHTML = '<i class="fa-solid fa-play"></i> Generate & Run';
        });
    });
