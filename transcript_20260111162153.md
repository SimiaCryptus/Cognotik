    // Tab Switching Logic
    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.classList.remove('active'));
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

        document.getElementById('tab-' + tabId).classList.add('active');
        event.target.classList.add('active');
    }

    function toggleSimView(view) {
        const renderEl = document.getElementById('sim-output-render');
        const logsEl = document.getElementById('sim-output-logs');
        
        if(view === 'render') {
            renderEl.style.display = 'block';
            logsEl.style.display = 'none';
        } else {
            renderEl.style.display = 'none';
            logsEl.style.display = 'block';
        }
    }

    // Simulator Logic
    function runSimulation() {
        const goal = document.getElementById('sim_goal').value;
        const platform = document.getElementById('sim_platform').value;
        const skill = document.getElementById('sim_skill').value;
        const steps = document.getElementById('sim_steps').value;
        const incValidation = document.getElementById('sim_validation').checked;
        const incTroubleshooting = document.getElementById('sim_troubleshooting').checked;

        const renderEl = document.getElementById('sim-output-render');
        const logsEl = document.getElementById('sim-output-logs');

        // Switch to logs view temporarily
        toggleSimView('logs');
        logsEl.innerHTML = '';

        const logs = [
            `[INFO] Initializing TutorialGenerationTask...`,
            `[INFO] Context: ${platform} | Level: ${skill}`,
            `[INFO] Analyzing goal: "${goal}"`,
            `[INFO] Generating outline (${steps} steps)...`,
            `[INFO] Drafting content...`,
            `[INFO] Injecting code snippets...`,
            `[INFO] Verifying technical accuracy...`,
            `[SUCCESS] Generation complete.`
        ];

        let logIndex = 0;
        const logInterval = setInterval(() => {
            if (logIndex < logs.length) {
                const line = document.createElement('div');
                line.textContent = `> ${new Date().toLocaleTimeString()} ${logs[logIndex]}`;
                line.style.marginBottom = '4px';
                if (logs[logIndex].includes('SUCCESS')) line.style.color = '#4ade80';
                else if (logs[logIndex].includes('INFO')) line.style.color = '#60a5fa';
                logsEl.appendChild(line);
                logsEl.scrollTop = logsEl.scrollHeight;
                logIndex++;
            } else {
                clearInterval(logInterval);
                setTimeout(() => {
                    generateMarkdown(goal, platform, skill, steps, incValidation, incTroubleshooting);
                    toggleSimView('render');
                }, 500);
            }
        }, 300);
    }

    function generateMarkdown(goal, platform, skill, steps, incValidation, incTroubleshooting) {
        const renderEl = document.getElementById('sim-output-render');
        
        let validationHTML = '';
        if (incValidation) {
            validationHTML = `
                <p><strong>Verify Success:</strong></p>
                <ol>
                    <li>Run <code>curl http://localhost:3000</code> to verify the build locally.</li>
                    <li>Check the console output for "Build Successful".</li>
                </ol>`;
        }

        let troubleshootingHTML = '';
        if (incTroubleshooting) {
            troubleshootingHTML = `
                <h3>Troubleshooting</h3>
                <p><strong>Error: Access Denied</strong></p>
                <p>Ensure your user has the correct permissions configured in the settings file.</p>
                <hr>`;
        }

        const html = `
            <h1>${goal}</h1>
            <p style="color: var(--text-dim); font-size: 0.9em;">
                <strong>Platform:</strong> ${platform} &bull; 
                <strong>Level:</strong> ${skill.charAt(0).toUpperCase() + skill.slice(1)}
            </p>
            <hr style="border-color: var(--border-subtle); margin: 1rem 0;">

            <h2>Prerequisites</h2>
            <ul>
                <li>Access to terminal/command line</li>
                <li>Administrator privileges</li>
            </ul>

            <h2>Step 1: Preparation</h2>
            <p>First, ensure your environment is up to date.</p>
            <pre><code>sudo apt-get update && sudo apt-get upgrade</code></pre>

            <h2>Step 2: Configuration</h2>
            <p>Create the necessary configuration files.</p>
            <pre><code>echo "config=true" > settings.ini</code></pre>
            ${validationHTML}

            <h2>Step 3: Execution</h2>
            <p>Run the main process to complete the task.</p>
            <pre><code>./deploy.sh --target production</code></pre>

            ${troubleshootingHTML}
