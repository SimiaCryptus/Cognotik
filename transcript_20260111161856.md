    function switchTab(tabId) {
        document.querySelectorAll('.tab-pane').forEach(el => el.classList.remove('active'));
        document.querySelectorAll('.tabs-nav button').forEach(el => el.classList.remove('active'));

    // Simulator Logic
    function runSimulation() {
        const thesis = document.getElementById('sim-thesis').value;
        const argCount = document.getElementById('sim-argCount').value;
        const useCounter = document.getElementById('sim-counter').checked;
        const useImages = document.getElementById('sim-images').checked;
        const logsEl = document.getElementById('sim-logs');
        
        logsEl.innerHTML = '';
        const steps = [
            `[INIT] Analyzing thesis: "${thesis.substring(0, 30)}..."`,
            useImages ? `[IMG] Generating cover art prompt...` : null,
            `[PLAN] Structuring outline...`,
            useImages ? `[IMG] Rendering outline visualization...` : null,
            `[TEXT] Drafting Introduction...`,
            `[TEXT] Developing Argument 1 (Logic + Evidence)...`,
            argCount > 1 ? `[TEXT] Developing Argument 2...` : null,
            argCount > 2 ? `[TEXT] Developing Argument 3...` : null,
            useCounter ? `[TEXT] Addressing counterarguments & rebuttals...` : null,
            `[TEXT] Drafting Conclusion...`,
            `[REV] Performing revision pass 1/1 (Tone Check)...`,
            `[FINAL] Assembling document...`,
            `[SUCCESS] Essay Generated.`
        ].filter(Boolean);
        
        let i = 0;
        const interval = setInterval(() => {
            if (i < steps.length) {
                const line = document.createElement('div');
                const timestamp = new Date().toLocaleTimeString('en-US', {hour12: false});
                line.textContent = `${timestamp} ${steps[i]}`;
                
                if (steps[i].includes('SUCCESS')) line.style.color = '#4ade80';
                else if (steps[i].includes('IMG')) line.style.color = '#f472b6';
                else if (steps[i].includes('REV')) line.style.color = '#fbbf24';
                else line.style.color = '#a3a3a3';
                
                logsEl.appendChild(line);
                logsEl.scrollTop = logsEl.scrollHeight;
                i++;
            } else {
                clearInterval(interval);
                const result = document.createElement('div');
                result.innerHTML = `
                    <br><strong style="color: #fff">Generated Structure:</strong><br>
                    1. <strong>Intro:</strong> Hook + Thesis Statement.<br>
                    2. <strong>Body:</strong> ${argCount} Core Arguments developed.<br>
                    ${useCounter ? '3. <strong>Rebuttal:</strong> Counter-argument addressed.<br>' : ''}
                    ${useCounter ? '4' : '3'}. <strong>Conclusion:</strong> Final synthesis and Call to Action.<br>
                    <br>
                    <em style="color: var(--accent-primary)">Output ready: essay.md, essay.pdf ${useImages ? ', 4 images' : ''}</em>
                `;
                logsEl.appendChild(result);
                logsEl.scrollTop = logsEl.scrollHeight;
            }
        }, 600);
    }
